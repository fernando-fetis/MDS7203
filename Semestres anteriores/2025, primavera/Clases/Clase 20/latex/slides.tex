\section{Recuerdo EBM}

\begin{frame}{Modelos basados en energía}
    Un EBM $p_\theta(x)\propto\exp\left(-E_\theta(x)\right)$ entrena $E_\theta:\R^D\to\R$ por máxima verosimilitud (aproximada) usando que:
    \begin{equation*}
        \nabla_\theta\log p_\theta(x)=\mathbb{E}_{p_\theta(x)}\left[\nabla_\theta E_\theta(x)\right]-\nabla_\theta E_\theta(x)
    \end{equation*}
    \vspace{-0.8cm}
    \begin{itemize}
        \item $\mathbb{E}_{p_\theta(x)}[\cdots]$ se estima usando Monte Carlo, donde las muestras desde $p_\theta(x)$ se generan usando \textbf{Langevin sampling}:
              \begin{equation*}
                  x_{t+1} = x_t+\frac{\epsilon}{2} \underbrace{\nabla_{x_k} \log p_\theta(x_k)}_{-\nabla_{x_k} E_\theta(x_k)}+\sqrt{\epsilon}\, z_k,
                  \quad\text{donde }
                  z_k\sim\mathcal{N}(0,\operatorname{I}_D).
              \end{equation*}
        \item Los gradientes $\nabla_\theta E_\theta(x)$ y $\nabla_x E_\theta(x)$ se calculan por diferenciación automática de la red neuronal.
    \end{itemize}
\end{frame}

\section{Score matching}

\begin{frame}{Motivación}
    En la recursión del sampling de Langevin,
    \begin{equation*}
        x_{t+1} = x_t+\frac{\epsilon}{2} \nabla_{x_k} \log p_\theta(x_k) + \sqrt{\epsilon}\, z_k,
    \end{equation*}
    es necesario computar $\nabla_x \log p_\theta(x)\in\R^D$ en cada paso de la iteración, lo que vuelve costoso el entrenamiento y la generación.

    Los \textbf{modelos basados en score} modelan directamente esta cantidad, denominada \textbf{score (de Stein)}, mediante una red neuronal $s_\theta:\R^D\to\R^D$ la cual es entrenada para que $s_\theta(x)\approx\nabla_x\log p_\theta(x)$.

    Esto evita la diferenciación automática en la dinámica de Langevin, volviendo más eficiente el muestreo tanto para entrenamiento como para generación.
\end{frame}

\begin{frame}{Divergencia de Fisher}
    Aprender directamente el vector $\nabla_x \log p(x)$ es posible ya que una función de score $s(x)=\nabla_x\log p(x)$ define una única función de densidad $p(x)$, y viceversa:
    \begin{equation*}
        \text{si}\quad
        \nabla_x \log p_1(x)=\nabla_x \log p_2(x)
        \quad\text{entonces}\quad
        p_1(x)=p_2(x)
    \end{equation*}
    Para entrenar un modelo basado en score, $s_\theta(x)=\nabla_x\log p_\theta(x)$ se puede utilizar la \textbf{divergencia de Fisher}:
    \begin{equation*}
        \operatorname{D_F}\left(p_{\text{data}}(x)\|p_\theta(x)\right):= \mathbb{E}_{p_{\text{data}}(x)}\left[\frac{1}{2}\left\|s_\theta(x)-\nabla_x\log p_{\text{data}}(x)\right\|^2\right]
    \end{equation*}
\end{frame}

\begin{frame}{Score matching}
    El score real $s_{\text{data}}(x)=\nabla_x\log p_{\text{data}}(x)$ es desconocido, por lo que la divergencia de Fisher no es computable directamente.

    Sin embargo, es posible reescribirla de forma tratable, dando paso a la técnica de \textbf{score matching}:
    \begin{equation*}
        \operatorname{D_F}\left(p_{\text{data}}(x)\|p_\theta(x)\right)= \mathbb{E}_{p_{\text{data}}(x)}\left[\frac{1}{2}\left\|s_\theta(x)\right\|^2 + \operatorname{Div}_x\left( s_\theta(x)\right)\right] + \operatorname{constante}
    \end{equation*}
    Por lo tanto, es posible entrenar $s_\theta(x)$ minimizando la expresión anterior, la cual no depende del score real $s_{\text{data}}(x)$.

    Para la generación de muestras, se puede usar Langevin sampling usando directamente $s_\theta(x)$ en vez de $\nabla_x\log p_\theta(x)$.
\end{frame}

\begin{frame}{Denoising score matching}
    La descomposición anterior involucra el cálculo de la divergencia $\operatorname{Div}_x(s_\theta(x))$, lo cual puede ser costoso.

    Un enfoque alternativo para que $\operatorname{D_F}\left(p_\theta(x)\|p_{\text{data}}(x)\right)$ sea tratable es utilizar otra distribución, $q_\sigma(\tilde x)$ que sea una versión levemente ruidosa de la distribución real $p_{\text{data}}(x)$.

    Eligiendo $q_\sigma(\tilde x| x)\sim \mathcal{N}(x,\sigma^2\operatorname{I}_D)$ (con $\sigma>0$ un hiperparámetro), su score tiene forma cerrada:
    \begin{equation*}
        \nabla_{\tilde x} \log q_\sigma(\tilde x|x)=  -\frac{\tilde x-x}{\sigma^2}
    \end{equation*}
    Esto es útil para otra posible descomposición de la divergencia de Fisher, dando paso a la técnica de \textbf{denoising score matching}.
\end{frame}

\begin{frame}{Denoising score matching}
    La divergencia de Fisher entre la distribución real alterada $q_\sigma(x)$ y el modelo paramétrico $p_\theta(x)$ se puede reescribir como:
    \begin{align*}
         & \operatorname{D_F}\left(q_\sigma(\tilde x)\|p_\theta(\tilde x)\right)                            \\
         & = \mathbb{E}_{p_{\text{data}}(x),\,q_\sigma(\tilde x|x)}\left[\frac{1}{2}\left\|
            s_\theta(\tilde x)-\nabla_{\tilde x} \log q_\sigma(\tilde x|x)
        \right\|^2\right] + \operatorname{constante}                                                        \\
         & =\mathbb{E}_{p_{\text{data}}(x),\,\epsilon\sim\mathcal{N}(0,\text{I}_D)}\left[\frac{1}{2}\left\|
            s_\theta(x+\sigma \epsilon)+\frac{\epsilon}{\sigma}
            \right\|^2\right] + \operatorname{constante}
    \end{align*}
    En consecuencia, se puede entrenar un modelo de score $s_\theta(x)$ para la distribución $q_\sigma(x)\approx p_{\text{data}}(x)$ de forma eficiente.

    Notar que el hiperparámetro $\sigma>0$ produce un trade-off entre el grado de similitud $q_\sigma(x)\approx p_{\text{data}}(x)$ y la varianza de $q_\sigma(\tilde x|x)$.
\end{frame}

\begin{frame}{DSM con varios niveles de ruido}
    El trade-off anterior puede ser evitado al considerar múltiples niveles de ruido en la técnica de DSM.

    Si $(\sigma_k)_{k=1}^K\subset\R_{++}$ es una secuencia creciente de ruidos, se pueden considerar distribuciones $q_{\sigma_k}(\tilde x| x)\sim \mathcal{N}(x,\sigma_k^2\operatorname{I}_D)$ para hacer DSM utilizando una única red neuronal $s_\theta:\R^D\times\R_{++}\to\R^D$ que minimice la función de pérdida combinada
    \begin{equation*}
        \mathcal{L}(\theta)
        =\sum_{k=1}^K \omega_k\,\mathbb{E}_{p_{\text{data}}(x),\,\epsilon\sim\mathcal{N}(0,\text{I}_D)}\left[\frac{1}{2}\left\|
            s_\theta(x+\sigma_k \epsilon)+\frac{\epsilon}{\sigma_k}
            \right\|^2\right],
    \end{equation*}
    donde $(\omega_k)_{k=1}^K\subset\R_{++}$ son ponderadores asignados a cada nivel de ruido.
\end{frame}

\begin{frame}{Annealed Langevin sampling}
    Para la generación de muestras desde el modelo anterior se puede generalizar el sampling de Langevin aplicándolo de forma secuencial de forma descendente en los niveles de ruido, y utilizando la muestra generada en el nivel $k+1$ como inicialización para el nivel $k$. Esto se denomina \textbf{annealed Langevin sampling}:

    \hspace{0.5cm}\vrule width 2pt\hspace{0.3cm}%
    \begin{minipage}[t]{0.85\textwidth}
        {\small
            \begin{itemize}
                \item[] \texttt{Inicializar} $x_{K+1}=0$
                \item[] \texttt{Para} $k\in\{K,\ldots,1\}$:
                \item[] \quad $\lambda_k = \lambda \left(\frac{\sigma_k}{\sigma_1}\right)^2$
                \item[] \quad  $x_{\sigma_k}\!=\!\text{Langevin}(\text{score\_fn}\!=\!s_\theta(\cdot,\sigma_k),\text{init}\!=\!x_{\sigma_{k+1}},\text{step}\!=\!\lambda_k)$
            \end{itemize}}
    \end{minipage}
\end{frame}
