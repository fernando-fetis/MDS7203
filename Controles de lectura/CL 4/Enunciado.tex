\section*{Redes generativas adversarias}
\begin{enumerate}[{\bf P1.}]
	\item ¿Cuál es la diferencia principal entre Pix2Pix y CycleGAN? ¿Qué ventaja tiene CycleGAN sobre Pix2Pix?

	\item ¿A qué se denomina colapso a la moda en el entrenamiento de una GAN? Explique por qué ocurre y cómo afecta a la distribución generada.

	\item La métrica FID (Fréchet Inception Distance) es el estándar para comparar modelos generativos de imágenes. Describa brevemente cómo se calcula: qué red neuronal se usa, qué se extrae de ella y cómo se comparan las distribuciones de datos reales y generados.
\end{enumerate}

\section*{Autoencoders variacionales}
\begin{enumerate}[{\bf P1.}]
	\item ¿Qué diferencia fundamental existe entre un autoencoder clásico y un autoencoder variacional (VAE)? En particular, explique cómo difieren en la naturaleza del espacio latente y en la forma en que se entrena cada modelo.

	\item La función objetivo de un VAE es la ELBO, cuya forma tratable es la siguiente:
	\begin{equation*}
		\operatorname{ELBO}(x)
		= \underbrace{\mathbb{E}_{q_\phi(z|x)}\!\left[\log p_\theta(x|z)\right]}_{\text{término de reconstrucción}}
		- \underbrace{\operatorname{D_{KL}}\!\left(q_\phi(z|x)\,\|\,p_\theta(z)\right)}_{\text{prior matching}}
	\end{equation*}
	Interprete cada uno de los dos términos.
\end{enumerate}
