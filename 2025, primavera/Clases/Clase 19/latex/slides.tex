\section{Modelos basados en energía}

\begin{frame}{Motivación}
    Parametrizar directamente una distribución $p_\theta(x)$ usando una red neuronal requiere imponer condiciones propias de una función de densidad:
    \begin{itemize}
        \item \textbf{Positividad:} $p_\theta(x)\geq 0$ para todo $x\in\R^D$.
        \item \textbf{Normalización:} $\int_{\R^D}p_\theta(x)\,\text{d}x=1$
    \end{itemize}
    La primera condición se puede imponer fácilmente usando funciones de activación como ReLU.

    En el caso discreto, la segunda condición se puede imponer usando la función softmax. Sin embargo, en el caso continuo esta cantidad es intratable de evaluar en general.
\end{frame}

\begin{frame}{Formulación}
    Los \textbf{modelos basados en energía (EBMs)} reparametrizan la distribución conjunta $p_\theta(x)$ como
    \begin{equation*}
        p_\theta(x)=\frac{\exp\left(-E_\theta(x)\right)}{Z_\theta},
    \end{equation*}
    donde $E_\theta:\R^D\to\R$  es aprendida por una red neuronal, mientras que $Z_\theta=\int_{\R^D} \exp\left(-E_\theta(x)\right)\text{d}x$ es la constante de normalización.
    \begin{itemize}
        \item La red neuronal $E_\theta$ (\textbf{función de energía}) ahora es irrestricta.
        \item La dificultad de traslada al cálculo de $Z_\theta$ (\textbf{función de partición}).
    \end{itemize}
    En consecuencia, la verosimilitud $\log p_\theta(x)$ sigue siendo intratable. Sin embargo, se verá una forma de entrenar $E_\theta$ por máxima verosimilitud (aproximada) sin calcular la constante $Z_\theta>0$.
\end{frame}

\begin{frame}{Entrenamiento por máxima verosimilitud}
    Calcular $\nabla_\theta \log p_\theta(x)$ requiere calcular $Z_\theta$, por lo que entrenar $E_\theta$ por máxima verosimilitud (exacta) es intratable en general.

    Sin embargo, este gradiente se puede \textbf{estimar} eficientemente En efecto, se tiene que:
    \begin{equation*}
        \nabla_\theta\log p_\theta(x)=\mathbb{E}_{p_\theta(x)}\left[\nabla_\theta E_\theta(x)\right]-\nabla_\theta E_\theta(x)
    \end{equation*}
    \vspace{-0.8cm}
    \begin{itemize}
        \item $\mathbb{E}_{p_\theta(x)}\left[\nabla_\theta E_\theta(x)\right]$ se estima usando estimaciones de Monte Carlo.
        \item $\nabla_\theta E_\theta(x)$ se calcula por diferenciación automática.
    \end{itemize}
    Por lo tanto, solo se necesita poder generar muestras desde $p_\theta(x)$ para poder entrenar la red neuronal $E_\theta$ y maximizar $\log p_\theta(x)$.
\end{frame}

\section{Langevin sampling}

\begin{frame}{Algoritmos MCMC}
    Se necesita poder generar muestras desde $p_\theta(x)=\frac{\exp\left(-E_\theta(x)\right)}{Z_\theta}$.

    Dado que $Z_\theta$ es intratable, es necesario utilizar métodos de \textbf{Markov chain Monte Carlo (MCMC)} para generar muestras desde $p_\theta(x)$.
    \begin{itemize}
        \item Este tipo de algoritmos permite generar muestras desde distribuciones donde solo se conoce una cantidad proporcional a la función de densidad.
        \item Este es el caso de los EBMs, donde $p_\theta(x)\propto\exp\left(-E_\theta(x)\right)$.
        \item El método MCMC más común en los EBMs es \textbf{Langevin sampling}.
    \end{itemize}
\end{frame}

\begin{frame}{Dinámica de Langevin}
    El algoritmo de Langevin comienza con una muestra $x_0\sim p(x_0)$ cualquiera (generada, p.g., desde una gaussiana o una uniforme). Luego, realiza la siguiente iteración hasta la convergencia:
    \begin{equation*}
        x_{t+1} = x_t+\frac{\epsilon}{2} \nabla_{x_k} \log p_\theta(x_k)+\sqrt{\epsilon}\, z_k,
        \quad\text{donde }
        z_k\sim\mathcal{N}(0,\operatorname{I}_D).
    \end{equation*}
    Este método puede ser visto como una especie de gradiente ascendente sobre $\log p_\theta(x)$, con un término adicional de ruido:
    \begin{itemize}
        \item La constante $\epsilon>0$ actúa como una tasa de aprendizaje.
        \item El ruido adicional $\sqrt{\epsilon}\, z_k$ evita que la recursión converja a una moda de la distribución $p_\theta(x)$.
    \end{itemize}
\end{frame}

\begin{frame}{Aplicación a un EBM}
    La iteración de Langevin sampling es:
    \begin{equation*}
        x_{t+1} = x_t+\frac{\epsilon}{2} \nabla_{x_k} \log p_\theta(x_k)+\sqrt{\epsilon}\, z_k,
        \quad\text{donde }
        z_k\sim\mathcal{N}(0,\operatorname{I}_D).
    \end{equation*}
    Notar que el gradiente $\nabla_{x} \log p_\theta(x)$ que necesita esta dinámica ahora es con respecto a $x$ y no con respecto a $\theta$.

    En particular, para un EBM $p_\theta(x)=\frac{\exp\left(-E_\theta(x)\right)}{Z_\theta}$, se puede obtener este gradiente por diferenciación automática:
    \begin{equation*}
        \nabla_x\log p_\theta(x) =-\nabla_xE_\theta(x)
    \end{equation*}
\end{frame}

\begin{frame}{Resumen EBMs}
    En resumen, un EBM parametriza $p_\theta(x)=\frac{\exp\left(-E_\theta(x)\right)}{Z_\theta}$ y es entrenado por máxima verosimilitud aproximada:
    \begin{itemize}
        \item $E_\theta:\R^D\to\R$ se entrena por gradiente ascendente usando que
              \begin{equation*}
                  \nabla_\theta\log p_\theta(x)=\mathbb{E}_{p_\theta(x)}\left[\nabla_\theta E_\theta(x)\right]-\nabla_\theta E_\theta(x)
              \end{equation*}
        \item La esperanza $\mathbb{E}_{p_\theta(x)}\left[\nabla_\theta E_\theta(x)\right]$ es estimada promediando $\nabla_\theta E_\theta(x)$ sobre muestras $x\sim p_\theta(x)$ usando Langevin Sampling:
              \begin{equation*}
                  x_{t+1} = x_t+\frac{\epsilon}{2} \nabla_{x_k} \log p_\theta(x_k)+\sqrt{\epsilon}\, z_k,
                  \quad\text{donde }
                  z_k\sim\mathcal{N}(0,\operatorname{I}_D).
              \end{equation*}
        \item El gradiente $\nabla_{x_k} \log p_\theta(x_k)$ se calcula por diferenciación automática:
              \begin{equation*}
                  \nabla_{x_k}\log p_\theta(x_k) =-\nabla_xE_\theta(x_k)
              \end{equation*}
    \end{itemize}
\end{frame}
