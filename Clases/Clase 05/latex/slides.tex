\section{Formulación de un LM}

\begin{frame}{Vocabulario}
    Si $\mathcal{V}_0=\{a_1,\ldots,a_K\}$ es un vocabulario común, los elementos de una secuencia $S=(w_1,\ldots,w_T)\in\mathcal{V}^*$ se denominan \textbf{tokens}.

    Es habitual considerar un vocabulario extendido, $\mathcal{V}\supset\mathcal{V}_0$, para incluir \textbf{tokens especiales}. Por ejemplo:
    \begin{itemize}
        \item \textbf{Begin of sentence:} $\text{<BOS>}$.
        \item \textbf{End of sentence:} $\text{<EOS>}$.
        \item \textbf{Unknown:} $\text{<UNK>}$.
        \item \textbf{Padding:} $\text{<PAD>}$.
    \end{itemize}
    En este caso:
    \begin{equation*}
        \mathcal{V}=\{a_1,\ldots,a_K,\underbrace{\text{<BOS>}}_{a_{K+1}},\underbrace{\text{<EOS>}}_{a_{K+2}},\underbrace{\text{<UNK>}}_{a_{K+3}},\underbrace{\text{<PAD>}}_{a_{K+4}}\}.
    \end{equation*}
\end{frame}

\begin{frame}{Corpus y conjunto de entrenamiento}
    \begin{itemize}
        \item Un \textbf{modelo de lenguaje} es un modelo generativo que asigna una probabilidad a cada secuencia de tokens $S\in\mathcal{V}^*$, es decir, aprende una distribución $p_\theta:\mathcal{V}^*\to[0,1]$.
        \item Para entrenar un LM, es necesario contar con un conjunto de entrenamiento $\mathcal{D}=\{S^1,\ldots,S^N\}\subset \mathcal{V}^*$.
        \item Dado un \textbf{corpus} con símbolos en $\mathcal{V}$, se puede construir $\mathcal{D}\subset\mathcal{V}^*$ considerando todas las subsecuencias que se puedan obtener a partir de dicho corpus.
        \item Las secuencias en $\mathcal{D}$ se suelen truncar o extender (usando $\text{<PAD>}$) a un mismo largo $T\in\N$. Esto permite entrenamiento en batches.
    \end{itemize}
\end{frame}

\begin{frame}{Formulación autorregresiva}
    Dada una secuencia de tokens, $S=(w_1,\ldots,w_T)\in\mathcal{V}^*$, un \textbf{modelo autorregresivo} descompone la probabilidad conjunta $p_\theta(S)$ de manera causal:
    \begin{equation*}
        p_\theta(w_1,\ldots,w_T)=p_\theta(w_1)\prod_{t=1}^{T-1} p_\theta(w_{t+1}|w_1,\ldots,w_t)
    \end{equation*}
    \begin{itemize}
        \item $w_1$ siempre es $\text{<BOS>}$, por lo que se puede fijar $p_\theta(w_1)=\delta_{\text{<BOS>}}(w_1)$.
        \item Por lo tanto, solo se requiere aprender $p_\theta(w_{t+1}|w_1,\ldots,w_t)$ para todo contexto $(w_1,\ldots, w_t)\in\mathcal{V}^*$ y posible continuación $w_{t+1}\in\mathcal{V}$.
        \item Con esta distribución aprendida, es posible generar nuevas secuencias de tokens mediante \textbf{ancestral sampling}.
    \end{itemize}
\end{frame}

\begin{frame}{Formulación autorregresiva}
    \begin{itemize}
        \item $p_\theta(w_{t+1}|w_1,\ldots,w_t)\sim\text{Categórica}\left(\mathcal{V};\lambda_\theta(w_1,\ldots,w_t)\right)$ es una distribución sobre el vocabulario $\mathcal{V}$, es decir:
              \begin{equation*}
                  p_\theta(w_{t+1}=a_k|w_1,\ldots,w_t)=\lambda_\theta(w_1,\ldots,w_t)_k\in[0,1]
              \end{equation*}
        \item El vector de probabilidades $\lambda_\theta(w_1,\ldots,w_t)\in\Delta^{|\mathcal{V}|}$ es aprendido por una red neuronal $\lambda_\theta:\mathcal{V}^*\to[0,1]^{|\mathcal{V}|}$.
        \item Esta red neuronal recibe un \textbf{contexto} $(w_1,\ldots,w_t)\in\mathcal{V}^*$ y entrega una distribución para el próximo token, $w_{t+1}\in\mathcal{V}$.
    \end{itemize}
\end{frame}

\section{Inferencia usando una RNN}

\begin{frame}{Vectores de embedding}
    \begin{itemize}
        \item Las redes neuronales esperan como entrada vectores de $\R^D$, por lo que es necesario obtener una \textbf{representación continua} para cada uno de los símbolos del vocabulario $\mathcal{V}$.
        \item Se debe asociar cada token $a\in\mathcal{V}$ con un \textbf{vector de embedding} $\operatorname{emb}(a)\in\R^D$.
        \item El operador $\operatorname{emb}:\mathcal{V}\to\R^D$ mapea el $k$-ésimo token de $\mathcal{V}$ a la $k$-ésima fila de una matriz $E\in\mathcal{M}_{|\mathcal{V}|,D}(\R)$.
        \item La matriz $E$ se llama \textbf{matriz de embedding} y es aprendida junto al resto de los parámetros de la red neuronal.
    \end{itemize}
\end{frame}

\begin{frame}{Redes neuronales recurrentes}
    \begin{itemize}
        \item Dado que la entrada $(w_1,\ldots,w_t)\in\mathcal{V}^*$ a la red neuronal es de largo variable, no es posible utilizar una red MLP estándar para parametrizar la distribución $p_\theta(w_{t+1}|w_1,\ldots,w_t)$.
        \item Una alternativa clásica para este tipo de problemas es usar una \textbf{red neuronal recurrente}. La red de Elman clásica procesa una secuencia de tokens $(w_1,\ldots,w_t)\in\mathcal{V}^*$ de forma recursiva:
              \begin{equation*}
                  \begin{cases}
                      h_0 = 0                                            \\
                      h_s =\tanh(A\operatorname{emb}(w_s)+Bh_{s-1}+d), &
                      1\leq s\leq t
                  \end{cases}
              \end{equation*}
        \item  De este modo, se puede considerar que
              \begin{equation*}
                  \lambda_\theta(w_1,\ldots,w_t)=\operatorname{softmax}(Ch_t+e)\in\Delta^{|\mathcal{V}|}
              \end{equation*}
    \end{itemize}
\end{frame}

\begin{frame}{Entrenamiento}
    \begin{itemize}
        \item  La red neuronal se puede entrenar maximizando la verosimilitud (normalizada) sobre un trainset $\mathcal{D}=\{S^1,\ldots,S^N\}\subset\mathcal{V}^*$:
              \begin{equation*}
                  \max_\theta \frac{1}{|\mathcal{D}|}\sum_{S\in\mathcal{D}} \frac{1}{|S|}\log p_\theta(S).
              \end{equation*}
        \item Las normalizaciones entregan independencia en el tamaño de batch y en los largos de las secuencias.
        \item Para una secuencia $S=(w_1,\ldots,w_T)\in\mathcal{V}^*$:
              \begin{align*}
                  \log p_\theta(S)
                  = & \log\left(p_\theta(w_1)\prod_{t=1}^{T-1} p_\theta(w_{t+1}|w_1,\ldots,w_t)\right) \\
                  = & \sum_{t=1}^{T-1} \log p_\theta(w_{t+1}|w_1,\ldots,w_t).
              \end{align*}
    \end{itemize}
\end{frame}

\begin{frame}{Generación}
    Se puede generar una nueva secuencia de tokens utilizando \textbf{ancestral sampling} en un LM entrenado:
    \begin{itemize}
        \item Se comienza con el token inicial $w_1=\text{<BOS>}$ y se evalúa el modelo neuronal para predecir el siguiente token.
        \item Esto se repite de manera iterativa usando como contexto la secuencia de tokens que ya han sido generados.
        \item La generación continúa hasta que se genera el token $\text{<EOS>}$ o hasta que se genera una cantidad máxima de tokens.
        \item Se puede realizar \textbf{generación condicional} comenzando con una secuencia de tokens iniciales (\textbf{prompt}), $S_\text{prompt}=(w_1,\ldots,w_t)$.
    \end{itemize}
\end{frame}

\begin{frame}{Heurísticas de generación}
    Se debe decidir cómo elegir el próximo token a partir de $p_\theta(w_{t+1}|w_1,\ldots,w_t)\sim\text{Categórica}\left(\mathcal{V};\lambda_\theta(w_1,\ldots,w_t)\right)$:
    \begin{itemize}
        \item \textbf{Sampling:} elegir $w_{t+1}$ sampleando desde $p_\theta(w_{t+1}|w_1,\ldots,w_t)$.
        \item \textbf{Greedy decoding:} elegir $w_{t+1}$ como el token más probable, i.e.,
              \begin{equation*}
                  w_{t+1}=a_k,\quad\text{donde}\quad
                  k=\operatornamewithlimits{arg\,max}_{k\in\{1,\ldots,|\mathcal{V}|\}} \lambda_\theta(w_1,\ldots,w_t)_k.
              \end{equation*}
        \item \textbf{Top-$K$:} si $\mathcal{I}=\{k_1,\ldots,k_K\}$ son las $K\in\N$ coordenadas de mayor valor del vector $\lambda_\theta(w_1,\ldots,w_t)\in\Delta^{|\mathcal{V}|}$, samplear usando
              \begin{equation*}
                  \tilde\lambda_\theta(w_1,\ldots,w_t)_k=
                  \begin{cases}
                      0                                                                                               & \text{si }k\not\in\mathcal{I} \\
                      \frac{\lambda_\theta(w_1,\ldots,w_t)_k}{\sum_{i\in\mathcal{I}}\lambda_\theta(w_1,\ldots,w_t)_i} & \text{si } k\in\mathcal{I}
                  \end{cases}.
              \end{equation*}
    \end{itemize}
\end{frame}

\begin{frame}{Heurísticas de generación}
    Si $l\in\R^{|\mathcal{V}|}$ es el vector de logits entregado por la red neuronal y $T>0$ es un parámetro de \textbf{temperatura}, se puede escalar el vector de logits $l\mapsto \frac{1}{T}l$ antes de aplicar la función softmax.

    El parámetro $T$ permite controlar el trade-off entre versatilidad y confianza en la generación:
    \begin{itemize}
        \item Si $T>1$, el vector de probabilidades se vuelve más uniforme.
        \item Si $T<1$, el vector de probabilidades se vuelve más determinista.
    \end{itemize}
    En particular, $T\to 0$ equivale a greedy decoding, mientras $T\to\infty$ equivale a un sampling uniforme sobre $\mathcal{V}$.
\end{frame}

\begin{frame}{Heurísticas de generación}
    Muchas veces la generación autorregresiva puede caer en ciclos de repetición. Para mitigar este problema, se puede aplicar una heurística de \textbf{penalización por repetición}.

    Si $c(a_k)\in\N$ es la cantidad de veces que el símbolo $a_k\in\mathcal{V}$ ha aparecido en la secuencia generada hasta el momento, se pueden escalar los logits $l\in\R^{|\mathcal{V}|}$ mediante
    \begin{equation*}
        \tilde{l}_k=
        \begin{cases}
            \frac{1}{\gamma^{c(a_k)}} \cdot l_k   & \text{si } l_k>0    \\
            \gamma^{c(a_k)} \cdot l_k & \text{si } l_k\leq 0
        \end{cases}
    \end{equation*}
    El hiperparámetro $\gamma\geq1$ controla el nivel de penalización.

    Si $\gamma>1$, entonces los logits de tokens repetidos decrecen, reduciendo su probabilidad tras aplicar softmax.
\end{frame}
