\section{Recuerdo clase anterior}

\begin{frame}{Modelos autorregresivos}
    Dada una secuencia de \textbf{tokens}, $S=(w_1,\ldots,w_T)\in\mathcal{V}^*$ sobre un \textbf{vocabulario} $\mathcal{V}$, un \textbf{LM} factoriza $p_\theta(S)$ de forma autorregresiva:
    \begin{equation*}
        p_\theta(w_1,\ldots,w_T)=p_\theta(w_1)\prod_{t=1}^{T-1} p_\theta(w_{t+1}|w_1,\ldots,w_t),
    \end{equation*}
    donde $p_\theta(w_{t+1}|w_1,\ldots,w_t)\sim\text{Categórica}\left(\mathcal{V};\lambda_\theta(w_1,\ldots,w_t)\right)$.

    El vector de probabilidades $\lambda_\theta(w_1,\ldots,w_t)\in\Delta^{|\mathcal{V}|}$ es aprendido por una red neuronal $\lambda_\theta:\mathcal{V}^*\to[0,1]^{|\mathcal{V}|}$ cuya entrada es de longitud variable.

    Esto último motivó a utilizar una \textbf{RNN} para modelar $\lambda_\theta$.
\end{frame}

\begin{frame}{Modelos autorregresivos}
    Dado un conjunto de entrenamiento $\mathcal{D}=\{S^1,\ldots,S^N\}\subset\mathcal{V}^*$, la red neuronal $\lambda_\theta$ se entrena optimizando
    \begin{equation*}
        \max_\theta \frac{1}{|\mathcal{D}|}\sum_{S\in\mathcal{D}} \frac{1}{|S|}\log p_\theta(S),
    \end{equation*}
    con $\log p_\theta(S)= \sum_{t=1}^{T-1} \log p_\theta(w_{t+1}|w_1,\ldots,w_t)$ si $S=(w_1,\ldots,w_T)$.

    Para la generación, se comienza con $w_1=\text{<BOS>}\in\mathcal{V}$, y predice el siguiente token usando el LM. Esto se repite hasta generar el token $\text{<EOS>}\in\mathcal{V}$ o hasta alcanzar una longitud máxima.
\end{frame}

\section{Arquitectura GPT}

\begin{frame}{Motivación}
    Las RNNs tienen limitaciones:
    \begin{itemize}
        \item Entrenamiento secuencial no paralelizable.
        \item Dificultad para retener información a largo plazo.
        \item Sesgos hacia el final de la secuencia.
    \end{itemize}
    La \textbf{arquitectura GPT} soluciona, al menos en principio, estas limitaciones:
    \begin{itemize}
        \item Permite procesar una secuencia completa de forma paralela.
        \item En todo momento es posible acceder a la información contenida en cualquier parte de la secuencia.
        \item No hay posiciones privilegiadas en la secuencia.
    \end{itemize}
\end{frame}

\begin{frame}{Idea general}
    La arquitectura GPT está compuesta por \textbf{bloques Transformer}, cada uno de los cuales tiene dos componentes principales:
    \begin{itemize}
        \item Un mecanismo de \textbf{self-attention} que permite, para cada token de la secuencia, obtener embeddings enriquecidos con información contextual de los otros tokens de la secuencia.
        \item Una \textbf{red feedforward} que procesa cada embedding contextual de forma individual para obtener un embedding final.
    \end{itemize}
    Esta arquitectura también incluye elementos adicionales como \textbf{positional encodings}, \textbf{capas de normalización} y \textbf{dropout}.
\end{frame}

\begin{frame}{Idea general}
    A grandes rasgos, la arquitectura GPT aplica los siguientes componentes en el orden indicado:
    \begin{enumerate}
        \item Capa de token embedding (con positional encoding).
        \item Bloques Transformers:
              \begin{itemize}
                  \item Self-attention (con layer norm y conexión residual).
                  \item Red feedforward (con layer norm y conexión residual).
              \end{itemize}
        \item Cabezal de lenguaje.
    \end{enumerate}
    Cada bloque Transformer puede verse como un módulo usado para \textit{mejorar} los embeddings iniciales de la secuencia.
\end{frame}

\begin{frame}{Layer normalization}
    La aplicación consecutiva de varias capas neuronales suele afectar el entrenamiento de la red (p.g. por \textbf{vanishing gradient} o \textbf{exploding gradient}).
    \begin{itemize}
        \item Este problema se puede aminorar utilizando \textbf{capas de normalización} entre los bloques de la red.
        \item En redes convolucionales es usual utilizar \textbf{batch normalization} o \textbf{group normalization}.
        \item No es obvio cómo aplicar batch norm en modelos secuenciales: las secuencias de tokens no suelen tener el mismo largo y muchas veces los batches de entrenamiento deben ser pequeños.
    \end{itemize}
\end{frame}

\begin{frame}{Layer normalization}
    Para una secuencia de embeddings, \textbf{layer normalization} normaliza cada vector de embedding de forma individual.

    Para un vector de embedding de la secuencia, $x\in\R^{D}$, primero se obtiene el embedding normalizado $\tilde x\in\R^D$, donde:
    \begin{equation*}
        \tilde x_d = \frac{x_d - \mu}{\sigma},
        \quad\text{para}\quad
        \mu = \frac{1}{D} \sum_{d=1}^{D} x_d
        \quad\text{y}\quad
        \sigma^2 = \frac{1}{D} \sum_{d=1}^{D} (x_d - \mu)^2
    \end{equation*}
    Luego, de forma análoga a lo realizado por batch norm, layer norm aplica una transformación coordenada a coordenada:
    \begin{equation*}
        \text{LayerNorm}(x) = \alpha \odot \tilde x + \beta,
    \end{equation*}
    donde $\alpha,\beta\in\R^D$ son parámetros entrenables.
\end{frame}

\begin{frame}{Self-attention}
    Dada una secuencia de embeddings, $(x_1,\ldots,x_T)\subset\R^D$:
    \begin{itemize}
        \item El objetivo del \textbf{mecanismo de self-attention} es obtener una secuencia de \textit{embeddings enriquecidos}, $(y_1,\ldots,y_T)\subset\R^P$.
        \item Cada \textbf{embedding contextual} $y_t\in\R^P$ combina el embedding $x_t\in\R^D$ con todos los otros vectores de embedding, $(x_1,\ldots,x_T)$, para poder incorporar información contextual.
        \item Esto soluciona el problema de memoria a largo plazo presente en las redes recurrentes.
    \end{itemize}
\end{frame}

\begin{frame}{Self-attention}
    \begin{itemize}
        \item Se comienza proyectando cada vector de embeddings de la secuencia $(x_1,\ldots,x_T)\subset\R^D$ a $\R^P$.
              Se necesitan 3 proyecciones:
              \begin{equation*}
                  q_t=W_Q\,x_t,
                  \qquad
                  k_t=W_K\,x_t,
                  \qquad
                  v_t=W_V\,x_t,
                  \quad\forall t\in\{1,\ldots,T\}.
              \end{equation*}
        \item $W_Q,W_K,W_V\in\mathcal{M}_{P,D}(\R)$ son parámetros entrenables.
        \item Cada embedding enriquecido, $y_i\in\R^P$, se calcula como
              \begin{equation*}
                  y_i=\sum_{j=1}^T a_{ij}\,v_j,
                  \quad\text{donde}\quad  a_{ij}\geq 0 \text{ y } \sum_{j=1}^T a_{ij}=1
              \end{equation*}
        \item $a_{ij}\in[0,1]$ representa la \textbf{atención} que coloca el token $i$ al token $j$. La elección usual es la \textbf{scaled dot-product attention}:
              \begin{equation*}
                  a_{ij} = \text{Softmax}\{s_{i1},\ldots,s_{iT}\}_j = \frac{e^{s_{ij}}}{\sum_{t=1}^T e^{s_{it}}},
                  \quad\text{donde}\quad
                  s_{ij}=\frac{\langle q_i,k_j\rangle}{\sqrt{P}}
              \end{equation*}
    \end{itemize}
\end{frame}

\begin{frame}{Self-attention}
    Para paralelizar el cálculo de la secuencia $(y_1,\ldots,y_T)\subset\R^P$ se deben guardar las proyecciones en matrices $Q,K,V\in \mathcal{M}_{T,P}(\R)$:
    \begin{equation*}
        Q=\left(
        \begin{matrix}
            q_1^\top \\
            \vdots   \\
            q_T^\top
        \end{matrix}
        \right),
        \qquad
        K=\left(
        \begin{matrix}
            k_1^\top \\
            \vdots   \\
            k_T^\top
        \end{matrix}
        \right),
        \qquad
        V=\left(
        \begin{matrix}
            v_1^\top \\
            \vdots   \\
            v_T^\top
        \end{matrix}
        \right)
    \end{equation*}
    Dado que $(QK^\top)_{ij}= \langle q_i,k_j\rangle$, entonces $s_{ij}=\left(\frac{QK^\top}{\sqrt{P}}\right)_{ij}$.

    Notando que $A=\text{softmax}_\text{por filas}\left(S\right)\in\mathcal{M}_{T,T}(\R)$, se tiene que:
    \begin{equation*}
        \left(
        \begin{matrix}
                y_1^\top \\
                \vdots   \\
                y_T^\top
            \end{matrix}
        \right)=\text{softmax}_\text{por filas}\left(\frac{QK^\top}{\sqrt{P}}\right)V
        \in\mathcal{M}_{T,P}(\R)
    \end{equation*}
\end{frame}

\begin{frame}{Masked self-attention}
    El mecanismo anterior pierde la \textbf{causalidad} del lenguaje al momento de computar los embeddings $(y_1,\ldots,y_T)\subset\R^P$:
    \begin{itemize}
        \item Cada embedding $y_t\in\R^P$ solo debería poder acceder a los embeddings $x_1,\ldots,x_t$ y no a los embeddings futuros, $x_{t+1},\ldots,x_T$.
        \item Para imponer esto, se pueden \textbf{enmascarar} los tokens futuros forzando que $a_{ij}=0$ para $j>i$.
    \end{itemize}
    Sin embargo, no se puede simplemente sustituir $a_{ij}=0$ para $j>i$ ya que se dejaría de cumplir la condición $\sum_{j=1}^T a_{ij}=1$.

    Lo usual es modificar $S\in\mathcal{M}_{T,T}(\R)$, sustituyendo cada entrada $s_{ij}$ (con $j>i$) por un valor muy negativo, de manera que $a_{ij}\approx 0$.
\end{frame}

\begin{frame}{Multihead self-attention}
    En las redes convolucionales es usual utilizar varios filtros convolucionales para capturar diferentes patrones.

    De forma análoga, en el mecanismo de self-attention se pueden utilizar múltiples \textbf{cabezales de atención} (MHSA) para capturar diferentes relaciones entre los tokens:
    \begin{itemize}
        \item Si $(x_1,\ldots,x_T)\subset\R^D$ es una secuencia de vectores de embeddings e $(y_1^{(h)},\ldots,y_T^{(h)})\subset\R^P$ es la salida del $h$-ésimo cabezal de atención, entonces se puede considerar embeddings
              \begin{equation*}
                  \tilde y_t=\left(y_t^{(1)}\,\cdots\, y_t^{(H)}\right)\in\R^{P\cdot H},\quad \text{para } t\in\{1,\ldots,T\}
              \end{equation*}

        \item Los embeddings finales $(u_1,\ldots,u_T)$ que retorna un MHSA son de la forma $u_t=W_0\, \tilde y_t\in\R^D$, donde la matriz $W_0\in\mathcal{M}_{D,\,P\cdot H}(\R)$ proyecta los embeddings $(\tilde y_1,\ldots,\tilde y_T)$ de vuelta al espacio $\R^D$.
    \end{itemize}
\end{frame}

\begin{frame}{Red feedforward}
    El último paso que realiza un bloque Transformer es pasar cada uno de los embeddings contextuales, $(u_1,\ldots,u_T)\subset\R^D$, por una misma MLP de dos capas. Para $t\in\{1,\ldots,T\}$:
    \begin{equation*}
        \text{FF}(u_t)=W_2\,\phi\left(W_1\,u_t\right),
        \quad\text{donde:}
    \end{equation*}
    \vspace{-0.8cm}
    \begin{itemize}
        \item $W_1\in\mathcal{M}_{J,D}(\R)$, $W_2\in\mathcal{M}_{D,J}(\R)$ son los parámetros de la red.
        \item  $\phi:\R\to\R$ es una función de activación que se aplica coordenada a coordenada (p.g., ReLU o GELU).
        \item Es usual considerar $J\gg D$ para darle más \textbf{capacidad} a la red.
    \end{itemize}
\end{frame}

\begin{frame}{Bloque Transformer}
    Almacenando la secuencia de embeddings, $(x_1,\ldots,x_T)\subset\R^D$, en las filas de una matriz $X\in\mathcal{M}_{T,D}(\R)$, un \textbf{bloque Transformer} tiene la siguiente estructura:
    \begin{equation*}
        \begin{cases}
            Y=X+\text{Dropout}_p(\text{MHSA}(\text{LayerNorm}(X))) \\
            \text{TransformerBlock}(X) = Y + \text{Dropout}_p(\text{FF}(\text{LayerNorm}(Y)))
        \end{cases}
    \end{equation*}
    \vspace{-0.5cm}
    \begin{itemize}
        \item Las capas de LN y la red FF se aplican de forma individual sobre cada vector de embedding (i.e., en cada fila de la matriz).
        \item Recordar que, para $x\in\R^D$, $\text{Dropout}_p(x):=\frac{1}{1-p}m\odot x$, donde la \textit{máscara} $m\in\{0,1\}^D$ es un vector con $m_d\sim\text{Bernoulli}(1-p)$.
    \end{itemize}
\end{frame}

\begin{frame}{Arquitectura GPT}
    Con esto, la arquitectura GPT realiza lo siguiente:
    \begin{itemize}
        \item Recibe una secuencia de tokens de entrada, $(w_1,\ldots,w_T)\in\mathcal{V}^*$.
        \item Esta secuencia es pasada por una capa de token embedding.
        \item Para inyectar la información posicional de cada token, se suma un vector de \textbf{positional embedding} a cada token embedding.
        \item Esta suma define los embeddings iniciales, los cuales luego son pasados por varios bloques Transformer consecutivos.
        \item Finalmente, se agrega un cabezal de lenguaje para proyectar los embedding finales a $\Delta^{|\mathcal{V}|}$.
    \end{itemize}
    En particular, se puede utilizar la salida del último token para predecir el siguiente token de la secuencia.
\end{frame}
